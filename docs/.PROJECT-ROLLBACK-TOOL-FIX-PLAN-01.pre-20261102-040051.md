**改为**
```js
const LIVE = ROOT
// R-01：演练根固定在 <ROOT>/.cache 之下，与 REPORT-14 §3.1 的声明一致（同时满足 R-03）
const RUNS_ROOT = path.join(ROOT, '.cache', 'fix14')

/** runId 解析：显式 > 唯一已存在 > 新建时间戳。多目录时拒绝（防误操作）。 */
function resolveRunDir() {
  const explicit = arg('run-id')
  if (explicit) return path.join(RUNS_ROOT, `run-${explicit}`)
  const existing = fs.existsSync(RUNS_ROOT)
    ? fs.readdirSync(RUNS_ROOT, { withFileTypes: true })
        .filter((e) => e.isDirectory() && /^run-/.test(e.name))
        .map((e) => e.name)
        .sort()
    : []
  if (existing.length > 1) {
    die(`存在多个演练目录（${existing.join(', ')}），请用 --run-id 指定其一`)
  }
  if (existing.length === 1) return path.join(RUNS_ROOT, existing[0])
  return path.join(RUNS_ROOT, `run-${new Date().toISOString().replace(/[:.]/g, '-')}`)
}

const DRILL = resolveRunDir()
```

### 6.2 改动：新增锁工具（建议插在 L142 之后，`need()` 之后）

```js
// ---------------- R-04：演练锁（绝不自动删） ----------------
const LOCK = path.join(DRILL, 'run.lock')

function lockInfo() {
  if (!fs.existsSync(LOCK)) return null
  try {
    return JSON.parse(fs.readFileSync(LOCK, 'utf8'))
  } catch (e) {
    return { unparsable: true, error: String(e && e.message) }
  }
}

function currentOperator() {
  const explicit = arg('operator')
  if (explicit) return explicit
  try {
    return os.userInfo().username
  } catch {
    return `unknown(pid=${process.pid})`
  }
}

/** 只创建，不删除。已存在即拒绝（含上次异常中断遗留）。 */
function acquireLock() {
  const existing = lockInfo()
  if (existing) {
    die(
      `已存在演练锁，拒绝执行：${LOCK}\n` +
        `${JSON.stringify(existing, null, 2)}\n` +
        `若已确认上一次演练结束或已人工恢复，请**人工删除**该文件后重试；本工具不会自动删锁。`
    )
  }
  const info = {
    pid: process.pid,
    startedAt: new Date().toISOString(),
    operator: currentOperator(),
    stage: STAGE,
    runDir: DRILL
  }
  fs.mkdirSync(DRILL, { recursive: true })
  const fd = fs.openSync(LOCK, 'wx')
  try {
    fs.writeSync(fd, JSON.stringify(info, null, 2) + '\n')
    fs.fsyncSync(fd)
  } finally {
    fs.closeSync(fd)
  }
  jwrite({ ev: 'lock-acquired', ...info })
}
```

**需要新增依赖**：`import os from 'node:os'`（插在 L22 之后）。

### 6.4 释放策略（**两子选项，需负责人择一**）
